export function findNearestBhopalLocality(lat: number, lng: number): BhopalLocality {
      let closest = BHOPAL_LOCALITIES[0];
        let minDist = Infinity;
          for (const loc of BHOPAL_LOCALITIES) {
                const d = (loc.lat - lat) ** 2 + (loc.lng - lng) ** 2;
                    if (d < minDist) {
                              minDist = d;
                                    closest = loc;
                    }
          }
            return closest;
}
